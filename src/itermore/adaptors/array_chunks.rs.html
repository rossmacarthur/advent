5/// An extension trait that provides the [`array_chunks`] method for iterators.
6///
7/// **Note:** the method provided here has a nightly API:
8/// [`Iterator::array_chunks`]. The nightly API will likely have better
9/// performance, so it should be preferred if possible.
10///
11/// [`array_chunks`]: IterArrayChunks::array_chunks
12pub trait IterArrayChunks: Iterator {
13    /// Returns an iterator over `N` elements of the iterator at a time.
14    ///
15    /// The chunks do not overlap. If `N` does not divide the length of the
16    /// iterator, then the last up to `N-1` elements will be omitted.
17    ///
18    /// # Panics
19    ///
20    /// If called with `N = 0`.
21    ///
22    /// # Examples
23    ///
24    /// Basic usage:
25    ///
26    /// ```
27    /// use itermore::IterArrayChunks;
28    ///
29    /// let mut iter = "lorem".chars().array_chunks();
30    /// assert_eq!(iter.next(), Some(['l', 'o']));
31    /// assert_eq!(iter.next(), Some(['r', 'e']));
32    /// assert_eq!(iter.next(), None);
33    /// ```
34    ///
35    /// ```
36    /// use itermore::IterArrayChunks;
37    ///
38    /// let data = [1, 1, 2, -2, 6, 0, 3, 1];
39    /// //          ^-----^  ^------^
40    /// for [x, y, z] in data.iter().array_chunks() {
41    ///     assert_eq!(x + y + z, 4);
42    /// }
43    /// ```
44    #[inline]
45    fn array_chunks<const N: usize>(self) -> ArrayChunks<Self, N>
46    where
47        Self: Sized,
48    {
49        ArrayChunks::new(self)
50    }
51
52    /// Identical to [`array_chunks`][IterArrayChunks::array_chunks] but doesn't
53    /// collide with the standard library name.
54    #[inline]
55    fn arrays<const N: usize>(self) -> ArrayChunks<Self, N>
56    where
57        Self: Sized,
58    {
59        ArrayChunks::new(self)
60    }
61}

63impl<I: ?Sized> IterArrayChunks for I where I: Iterator {}

65/// An iterator over `N` elements of the iterator at a time.
66///
67/// This struct is created by the [`array_chunks`] method on iterators. See its
68/// documentation for more.
69///
70/// [`array_chunks`]: IterArrayChunks::array_chunks
71#[derive(Debug, Clone)]
72#[must_use = "iterators are lazy and do nothing unless consumed"]
73pub struct ArrayChunks<I, const N: usize>
74where
75    I: Iterator,
76{
77    iter: I,
78    remainder: Option<IntoIter<I::Item, N>>,
79}
80
81impl<I, const N: usize> ArrayChunks<I, N>
82where
83    I: Iterator,
84{
85    #[track_caller]
86    fn new(iter: I) -> Self {
87        assert!(N != 0, "chunk size must be non-zero");
88        Self {
89            iter,
90            remainder: None,
91        }
92    }
93
94    /// Returns an iterator over the remaining elements of the original iterator
95    /// that are not going to be yielded. The returned iterator will yield at
96    /// most `N-1` elements.
97    #[inline]
98    pub fn into_remainder(mut self) -> IntoIter<I::Item, N> {
99        if self.remainder.is_none() {
100            self.by_ref().for_each(drop);
101        }
102        self.remainder.unwrap()
103    }
104}
105
106impl<I: Iterator, const N: usize> Iterator for ArrayChunks<I, N>
107where
108    I: Iterator,
109{
110    type Item = [I::Item; N];
111
112    #[inline]
113    fn next(&mut self) -> Option<Self::Item> {
114        let Self { iter, remainder } = self;
115        match arrays::from_iter(iter) {
116            Ok(chunk) => Some(chunk),
117            Err(rem) => {
118                remainder.get_or_insert(rem);
119                None
120            }
121        }
122    }
123
124    #[inline]
125    fn size_hint(&self) -> (usize, Option<usize>) {
126        let (lower, upper) = self.iter.size_hint();
127        (lower / N, upper.map(|n| n / N))
128    }
129
130    #[inline]
131    fn count(self) -> usize {
132        self.iter.count() / N
133    }
134}

136impl<I, const N: usize> DoubleEndedIterator for ArrayChunks<I, N>
137where
138    I: DoubleEndedIterator + ExactSizeIterator,
139{
140    #[inline]
141    fn next_back(&mut self) -> Option<Self::Item> {
142        self.next_back_remainder();
143        let mut rev = self.iter.by_ref().rev();
144        let mut chunk = arrays::from_iter(&mut rev).ok()?;
145        chunk.reverse();
146        Some(chunk)
147    }
148}

150impl<I, const N: usize> ExactSizeIterator for ArrayChunks<I, N>
151where
152    I: ExactSizeIterator,
153{
154    #[inline]
155    fn len(&self) -> usize {
156        self.iter.len() / N
157    }
158}

160impl<I, const N: usize> ArrayChunks<I, N>
161where
162    I: DoubleEndedIterator + ExactSizeIterator,
163{
164    /// Updates `self.remainder` such that `self.iter.len` is divisible by `N`.
165    #[inline]
166    fn next_back_remainder(&mut self) {
167        if self.remainder.is_some() {
168            return;
169        }
170
171        let rem = self.iter.len() % N;
172        let mut rev = self.iter.by_ref().rev().take(rem);
173        // SAFETY: `unwrap_err` always succeeds because x % N < N for all x.
174        let mut remainder = unsafe { arrays::from_iter(&mut rev).unwrap_err_unchecked() };
175
176        // We used `.rev()` above, so we need to re-reverse the remainder.
177        remainder.as_mut_slice().reverse();
178        self.remainder = Some(remainder);
179    }
180}
