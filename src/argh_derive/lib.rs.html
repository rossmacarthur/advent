29/// Entrypoint for `#[derive(FromArgs)]`.
30#[proc_macro_derive(FromArgs, attributes(argh))]
31pub fn argh_derive(input: proc_macro::TokenStream) -> proc_macro::TokenStream {
32    let ast = syn::parse_macro_input!(input as syn::DeriveInput);
33    let gen = impl_from_args(&ast);
34    gen.into()
35}

37/// Entrypoint for `#[derive(ArgsInfo)]`.
38#[proc_macro_derive(ArgsInfo, attributes(argh))]
39pub fn args_info_derive(input: proc_macro::TokenStream) -> proc_macro::TokenStream {
40    let ast = syn::parse_macro_input!(input as syn::DeriveInput);
41    let gen = args_info::impl_args_info(&ast);
42    gen.into()
43}

45/// Transform the input into a token stream containing any generated implementations,
46/// as well as all errors that occurred.
47fn impl_from_args(input: &syn::DeriveInput) -> TokenStream {
48    let errors = &Errors::default();
49    let type_attrs = &TypeAttrs::parse(errors, input);
50    let mut output_tokens = match &input.data {
51        syn::Data::Struct(ds) => {
52            impl_from_args_struct(errors, &input.ident, type_attrs, &input.generics, ds)
53        }
54        syn::Data::Enum(de) => {
55            impl_from_args_enum(errors, &input.ident, type_attrs, &input.generics, de)
56        }
57        syn::Data::Union(_) => {
58            errors.err(input, "`#[derive(FromArgs)]` cannot be applied to unions");
59            TokenStream::new()
60        }
61    };
62    errors.to_tokens(&mut output_tokens);
63    output_tokens
64}

66/// The kind of optionality a parameter has.
67enum Optionality {
68    None,
69    Defaulted(TokenStream),
70    Optional,
71    Repeating,
72    DefaultedRepeating(TokenStream),
73}
74
75impl PartialEq<Optionality> for Optionality {
76    fn eq(&self, other: &Optionality) -> bool {
77        use Optionality::*;
78        // NB: (Defaulted, Defaulted) can't contain the same token streams
79        matches!((self, other), (Optional, Optional) | (Repeating, Repeating))
80    }
81}
82
83impl Optionality {
84    /// Whether or not this is `Optionality::None`
85    fn is_required(&self) -> bool {
86        matches!(self, Optionality::None)
87    }
88}

90/// A field of a `#![derive(FromArgs)]` struct with attributes and some other
91/// notable metadata appended.
92struct StructField<'a> {
93    /// The original parsed field
94    field: &'a syn::Field,
95    /// The parsed attributes of the field
96    attrs: FieldAttrs,
97    /// The field name. This is contained optionally inside `field`,
98    /// but is duplicated non-optionally here to indicate that all field that
99    /// have reached this point must have a field name, and it no longer
100    /// needs to be unwrapped.
101    name: &'a syn::Ident,
102    /// Similar to `name` above, this is contained optionally inside `FieldAttrs`,
103    /// but here is fully present to indicate that we only have to consider fields
104    /// with a valid `kind` at this point.
105    kind: FieldKind,
106    // If `field.ty` is `Vec<T>` or `Option<T>`, this is `T`, otherwise it's `&field.ty`.
107    // This is used to enable consistent parsing code between optional and non-optional
108    // keyed and subcommand fields.
109    ty_without_wrapper: &'a syn::Type,
110    // Whether the field represents an optional value, such as an `Option` subcommand field
111    // or an `Option` or `Vec` keyed argument, or if it has a `default`.
112    optionality: Optionality,
113    // The `--`-prefixed name of the option, if one exists.
114    long_name: Option<String>,
115}
116
117impl<'a> StructField<'a> {
118    /// Attempts to parse a field of a `#[derive(FromArgs)]` struct, pulling out the
119    /// fields required for code generation.
120    fn new(errors: &Errors, field: &'a syn::Field, attrs: FieldAttrs) -> Option<Self> {
121        let name = field.ident.as_ref().expect("missing ident for named field");
122
123        // Ensure that one "kind" is present (switch, option, subcommand, positional)
124        let kind = if let Some(field_type) = &attrs.field_type {
125            field_type.kind
126        } else {
127            errors.err(
128                field,
129                concat!(
130                    "Missing `argh` field kind attribute.\n",
131                    "Expected one of: `switch`, `option`, `remaining`, `subcommand`, `positional`",
132                ),
133            );
134            return None;
135        };
136
137        // Parse out whether a field is optional (`Option` or `Vec`).
138        let optionality;
139        let ty_without_wrapper;
140        match kind {
141            FieldKind::Switch => {
142                if !ty_expect_switch(errors, &field.ty) {
143                    return None;
144                }
145                optionality = Optionality::Optional;
146                ty_without_wrapper = &field.ty;
147            }
148            FieldKind::Option | FieldKind::Positional => {
149                if let Some(default) = &attrs.default {
150                    let tokens = match TokenStream::from_str(&default.value()) {
151                        Ok(tokens) => tokens,
152                        Err(_) => {
153                            errors.err(&default, "Invalid tokens: unable to lex `default` value");
154                            return None;
155                        }
156                    };
157                    // Set the span of the generated tokens to the string literal
158                    let tokens: TokenStream = tokens
159                        .into_iter()
160                        .map(|mut tree| {
161                            tree.set_span(default.span());
162                            tree
163                        })
164                        .collect();
165                    let inner = if let Some(x) = ty_inner(&["Vec"], &field.ty) {
166                        optionality = Optionality::DefaultedRepeating(tokens);
167                        x
168                    } else {
169                        optionality = Optionality::Defaulted(tokens);
170                        &field.ty
171                    };
172                    ty_without_wrapper = inner;
173                } else {
174                    let mut inner = None;
175                    optionality = if let Some(x) = ty_inner(&["Option"], &field.ty) {
176                        inner = Some(x);
177                        Optionality::Optional
178                    } else if let Some(x) = ty_inner(&["Vec"], &field.ty) {
179                        inner = Some(x);
180                        Optionality::Repeating
181                    } else {
182                        Optionality::None
183                    };
184                    ty_without_wrapper = inner.unwrap_or(&field.ty);
185                }
186            }
187            FieldKind::SubCommand => {
188                let inner = ty_inner(&["Option"], &field.ty);
189                optionality =
190                    if inner.is_some() { Optionality::Optional } else { Optionality::None };
191                ty_without_wrapper = inner.unwrap_or(&field.ty);
192            }
193        }
194
195        // Determine the "long" name of options and switches.
196        // Defaults to the kebab-case'd field name if `#[argh(long = "...")]` is omitted.
197        let long_name = match kind {
198            FieldKind::Switch | FieldKind::Option => {
199                let long_name = attrs.long.as_ref().map(syn::LitStr::value).unwrap_or_else(|| {
200                    let kebab_name = to_kebab_case(&name.unraw().to_string());
201                    check_long_name(errors, name, &kebab_name);
202                    kebab_name
203                });
204                if long_name == "help" {
205                    errors.err(field, "Custom `--help` flags are not supported.");
206                }
207                let long_name = format!("--{}", long_name);
208                Some(long_name)
209            }
210            FieldKind::SubCommand | FieldKind::Positional => None,
211        };
212
213        Some(StructField { field, attrs, kind, optionality, ty_without_wrapper, name, long_name })
214    }
215
216    pub(crate) fn positional_arg_name(&self) -> String {
217        self.attrs
218            .arg_name
219            .as_ref()
220            .map(LitStr::value)
221            .unwrap_or_else(|| self.name.to_string().trim_matches('_').to_owned())
222    }
223}

225fn to_kebab_case(s: &str) -> String {
226    let words = s.split('_').filter(|word| !word.is_empty());
227    let mut res = String::with_capacity(s.len());
228    for word in words {
229        if !res.is_empty() {
230            res.push('-')
231        }
232        res.push_str(word)
233    }
234    res
235}

243    }

248    check("foo_bar", "foo-bar");

253/// Implements `FromArgs` and `TopLevelCommand` or `SubCommand` for a `#[derive(FromArgs)]` struct.
254fn impl_from_args_struct(
255    errors: &Errors,
256    name: &syn::Ident,
257    type_attrs: &TypeAttrs,
258    generic_args: &syn::Generics,
259    ds: &syn::DataStruct,
260) -> TokenStream {
261    let fields = match &ds.fields {
262        syn::Fields::Named(fields) => fields,
263        syn::Fields::Unnamed(_) => {
264            errors.err(
265                &ds.struct_token,
266                "`#![derive(FromArgs)]` is not currently supported on tuple structs",
267            );
268            return TokenStream::new();
269        }
270        syn::Fields::Unit => {
271            errors.err(&ds.struct_token, "#![derive(FromArgs)]` cannot be applied to unit structs");
272            return TokenStream::new();
273        }
274    };
275
276    let fields: Vec<_> = fields
277        .named
278        .iter()
279        .filter_map(|field| {
280            let attrs = FieldAttrs::parse(errors, field);
281            StructField::new(errors, field, attrs)
282        })
283        .collect();
284
285    ensure_unique_names(errors, &fields);
286    ensure_only_last_positional_is_optional(errors, &fields);
287
288    let impl_span = Span::call_site();
289
290    let from_args_method = impl_from_args_struct_from_args(errors, type_attrs, &fields);
291
292    let redact_arg_values_method =
293        impl_from_args_struct_redact_arg_values(errors, type_attrs, &fields);
294
295    let top_or_sub_cmd_impl = top_or_sub_cmd_impl(errors, name, type_attrs, generic_args);
296
297    let (impl_generics, ty_generics, where_clause) = generic_args.split_for_impl();
298    let trait_impl = quote_spanned! { impl_span =>
299        #[automatically_derived]
300        impl #impl_generics argh::FromArgs for #name #ty_generics #where_clause {
301            #from_args_method
302
303            #redact_arg_values_method
304        }
305
306        #top_or_sub_cmd_impl
307    };
308
309    trait_impl
310}

312fn impl_from_args_struct_from_args<'a>(
313    errors: &Errors,
314    type_attrs: &TypeAttrs,
315    fields: &'a [StructField<'a>],
316) -> TokenStream {
317    let init_fields = declare_local_storage_for_from_args_fields(fields);
318    let unwrap_fields = unwrap_from_args_fields(fields);
319    let positional_fields: Vec<&StructField<'_>> =
320        fields.iter().filter(|field| field.kind == FieldKind::Positional).collect();
321    let positional_field_idents = positional_fields.iter().map(|field| &field.field.ident);
322    let positional_field_names = positional_fields.iter().map(|field| field.name.to_string());
323    let last_positional_is_repeating = positional_fields
324        .last()
325        .map(|field| field.optionality == Optionality::Repeating)
326        .unwrap_or(false);
327    let last_positional_is_greedy = positional_fields
328        .last()
329        .map(|field| field.kind == FieldKind::Positional && field.attrs.greedy.is_some())
330        .unwrap_or(false);
331
332    let flag_output_table = fields.iter().filter_map(|field| {
333        let field_name = &field.field.ident;
334        match field.kind {
335            FieldKind::Option => Some(quote! { argh::ParseStructOption::Value(&mut #field_name) }),
336            FieldKind::Switch => Some(quote! { argh::ParseStructOption::Flag(&mut #field_name) }),
337            FieldKind::SubCommand | FieldKind::Positional => None,
338        }
339    });
340
341    let flag_str_to_output_table_map = flag_str_to_output_table_map_entries(fields);
342
343    let mut subcommands_iter =
344        fields.iter().filter(|field| field.kind == FieldKind::SubCommand).fuse();
345
346    let subcommand: Option<&StructField<'_>> = subcommands_iter.next();
347    for dup_subcommand in subcommands_iter {
348        errors.duplicate_attrs("subcommand", subcommand.unwrap().field, dup_subcommand.field);
349    }
350
351    let impl_span = Span::call_site();
352
353    let missing_requirements_ident = syn::Ident::new("__missing_requirements", impl_span);
354
355    let append_missing_requirements =
356        append_missing_requirements(&missing_requirements_ident, fields);
357
358    let parse_subcommands = if let Some(subcommand) = subcommand {
359        let name = subcommand.name;
360        let ty = subcommand.ty_without_wrapper;
361        quote_spanned! { impl_span =>
362            Some(argh::ParseStructSubCommand {
363                subcommands: <#ty as argh::SubCommands>::COMMANDS,
364                dynamic_subcommands: &<#ty as argh::SubCommands>::dynamic_commands(),
365                parse_func: &mut |__command, __remaining_args| {
366                    #name = Some(<#ty as argh::FromArgs>::from_args(__command, __remaining_args)?);
367                    Ok(())
368                },
369            })
370        }
371    } else {
372        quote_spanned! { impl_span => None }
373    };
374
375    let help_triggers = get_help_triggers(type_attrs);
376
377    let help = if cfg!(feature = "help") {
378        // Identifier referring to a value containing the name of the current command as an `&[&str]`.
379        let cmd_name_str_array_ident = syn::Ident::new("__cmd_name", impl_span);
380        help::help(errors, cmd_name_str_array_ident, type_attrs, fields, subcommand, &help_triggers)
381    } else {
382        quote! { String::new() }
383    };
384
385    let method_impl = quote_spanned! { impl_span =>
386        fn from_args(__cmd_name: &[&str], __args: &[&str])
387            -> std::result::Result<Self, argh::EarlyExit>
388        {
389            #![allow(clippy::unwrap_in_result)]
390
391            #( #init_fields )*
392
393            argh::parse_struct_args(
394                __cmd_name,
395                __args,
396                argh::ParseStructOptions {
397                    arg_to_slot: &[ #( #flag_str_to_output_table_map ,)* ],
398                    slots: &mut [ #( #flag_output_table, )* ],
399                    help_triggers: &[ #( #help_triggers ),* ],
400                },
401                argh::ParseStructPositionals {
402                    positionals: &mut [
403                        #(
404                            argh::ParseStructPositional {
405                                name: #positional_field_names,
406                                slot: &mut #positional_field_idents as &mut argh::ParseValueSlot,
407                            },
408                        )*
409                    ],
410                    last_is_repeating: #last_positional_is_repeating,
411                    last_is_greedy: #last_positional_is_greedy,
412                },
413                #parse_subcommands,
414                &|| #help,
415            )?;
416
417            let mut #missing_requirements_ident = argh::MissingRequirements::default();
418            #(
419                #append_missing_requirements
420            )*
421            #missing_requirements_ident.err_on_any()?;
422
423            Ok(Self {
424                #( #unwrap_fields, )*
425            })
426        }
427    };
428
429    method_impl
430}

432/// get help triggers vector from type_attrs.help_triggers as a [`Vec<String>`]
433///
434/// Defaults to vec!["--help", "help"] if type_attrs.help_triggers is None
435fn get_help_triggers(type_attrs: &TypeAttrs) -> Vec<String> {
436    let help_triggers = type_attrs.help_triggers.as_ref().map_or_else(
437        || vec!["--help".to_owned(), "help".to_owned()],
438        |s| {
439            s.iter()
440                .filter_map(|s| {
441                    let trigger = s.value();
442                    let trigger_trimmed = trigger.trim().to_owned();
443                    if trigger_trimmed.is_empty() {
444                        None
445                    } else {
446                        Some(trigger_trimmed)
447                    }
448                })
449                .collect::<Vec<_>>()
450        },
451    );
452    help_triggers
453}

536        fn redact_arg_values(__cmd_name: &[&str], __args: &[&str]) -> std::result::Result<Vec<String>, argh::EarlyExit> {
537            #( #init_fields )*
538
539            argh::parse_struct_args(
540                __cmd_name,
541                __args,
542                argh::ParseStructOptions {
543                    arg_to_slot: &[ #( #flag_str_to_output_table_map ,)* ],

546                },
547                argh::ParseStructPositionals {
548                    positionals: &mut [
549                        #(
550                            argh::ParseStructPositional {
551                                name: #positional_field_names,
552                                slot: &mut #positional_field_idents as &mut argh::ParseValueSlot,
553                            },
554                        )*
555                    ],
556                    last_is_repeating: #last_positional_is_repeating,
557                    last_is_greedy: #last_positional_is_greedy,
558                },
559                #redact_subcommands,
560                &|| #help,
561            )?;
562
563            let mut #missing_requirements_ident = argh::MissingRequirements::default();
564            #(
565                #append_missing_requirements
566            )*
567            #missing_requirements_ident.err_on_any()?;
568
569            let mut __redacted = vec![
570                if let Some(cmd_name) = __cmd_name.last() {
571                    (*cmd_name).to_owned()
572                } else {
573                    return Err(argh::EarlyExit::from(#unwrap_cmd_name_err_string.to_owned()));
574                }
575            ];
576
577            #( #unwrap_fields )*
578
579            Ok(__redacted)
580        }
581    };
582
583    method_impl
584}

586/// Ensures that only the last positional arg is non-required.
587fn ensure_only_last_positional_is_optional(errors: &Errors, fields: &[StructField<'_>]) {
588    let mut first_non_required_span = None;
589    for field in fields {
590        if field.kind == FieldKind::Positional {
591            if let Some(first) = first_non_required_span {
592                errors.err_span(
593                    first,
594                    "Only the last positional argument may be `Option`, `Vec`, or defaulted.",
595                );
596                errors.err(&field.field, "Later positional argument declared here.");
597                return;
598            }
599            if !field.optionality.is_required() {
600                first_non_required_span = Some(field.field.span());
601            }
602        }
603    }
604}

606/// Ensures that only one short or long name is used.
607fn ensure_unique_names(errors: &Errors, fields: &[StructField<'_>]) {
608    let mut seen_short_names = HashMap::new();
609    let mut seen_long_names = HashMap::new();
610
611    for field in fields {
612        if let Some(short_name) = &field.attrs.short {
613            let short_name = short_name.value();
614            if let Some(first_use_field) = seen_short_names.get(&short_name) {
615                errors.err_span_tokens(
616                    first_use_field,
617                    &format!("The short name of \"-{}\" was already used here.", short_name),
618                );
619                errors.err_span_tokens(field.field, "Later usage here.");
620            }
621
622            seen_short_names.insert(short_name, &field.field);
623        }
624
625        if let Some(long_name) = &field.long_name {
626            if let Some(first_use_field) = seen_long_names.get(&long_name) {
627                errors.err_span_tokens(
628                    *first_use_field,
629                    &format!("The long name of \"{}\" was already used here.", long_name),
630                );
631                errors.err_span_tokens(field.field, "Later usage here.");
632            }
633
634            seen_long_names.insert(long_name, field.field);
635        }
636    }

639/// Implement `argh::TopLevelCommand` or `argh::SubCommand` as appropriate.
640fn top_or_sub_cmd_impl(
641    errors: &Errors,
642    name: &syn::Ident,
643    type_attrs: &TypeAttrs,
644    generic_args: &syn::Generics,
645) -> TokenStream {
646    let description = if cfg!(feature = "help") {
647        help::require_description(errors, name.span(), &type_attrs.description, "type")
648    } else {
649        String::new()
650    };
651    let (impl_generics, ty_generics, where_clause) = generic_args.split_for_impl();
652    if type_attrs.is_subcommand.is_none() {
653        // Not a subcommand
654        quote! {
655            #[automatically_derived]
656            impl #impl_generics argh::TopLevelCommand for #name #ty_generics #where_clause {}
657        }
658    } else {
659        let empty_str = syn::LitStr::new("", Span::call_site());
660        let subcommand_name = type_attrs.name.as_ref().unwrap_or_else(|| {
661            errors.err(name, "`#[argh(name = \"...\")]` attribute is required for subcommands");
662            &empty_str
663        });
664        quote! {
665            #[automatically_derived]
666            impl #impl_generics argh::SubCommand for #name #ty_generics #where_clause {
667                const COMMAND: &'static argh::CommandInfo = &argh::CommandInfo {
668                    name: #subcommand_name,
669                    description: #description,
670                };
671            }
672        }
673    }
674}

676/// Declare a local slots to store each field in during parsing.
677///
678/// Most fields are stored in `Option<FieldType>` locals.
679/// `argh(option)` fields are stored in a `ParseValueSlotTy` along with a
680/// function that knows how to decode the appropriate value.
681fn declare_local_storage_for_from_args_fields<'a>(
682    fields: &'a [StructField<'a>],
683) -> impl Iterator<Item = TokenStream> + 'a {
684    fields.iter().map(|field| {
685        let field_name = &field.field.ident;
686        let field_type = &field.ty_without_wrapper;
687
688        // Wrap field types in `Option` if they aren't already `Option` or `Vec`-wrapped.
689        let field_slot_type = match field.optionality {
690            Optionality::Optional | Optionality::Repeating => (&field.field.ty).into_token_stream(),
691            Optionality::None | Optionality::Defaulted(_) => {
692                quote! { std::option::Option<#field_type> }
693            }
694            Optionality::DefaultedRepeating(_) => {
695                quote! { std::option::Option<std::vec::Vec<#field_type>> }
696            }
697        };
698
699        match field.kind {
700            FieldKind::Option | FieldKind::Positional => {
701                let from_str_fn = match &field.attrs.from_str_fn {
702                    Some(from_str_fn) => from_str_fn.into_token_stream(),
703                    None => {
704                        quote! {
705                            <#field_type as argh::FromArgValue>::from_arg_value
706                        }
707                    }
708                };
709
710                quote! {
711                    let mut #field_name: argh::ParseValueSlotTy<#field_slot_type, #field_type>
712                        = argh::ParseValueSlotTy {
713                            slot: std::default::Default::default(),
714                            parse_func: |_, value| { #from_str_fn(value) },
715                        };
716                }
717            }
718            FieldKind::SubCommand => {
719                quote! { let mut #field_name: #field_slot_type = None; }
720            }
721            FieldKind::Switch => {
722                quote! { let mut #field_name: #field_slot_type = argh::Flag::default(); }
723            }
724        }
725    })
726}

728/// Unwrap non-optional fields and take options out of their tuple slots.
729fn unwrap_from_args_fields<'a>(
730    fields: &'a [StructField<'a>],
731) -> impl Iterator<Item = TokenStream> + 'a {
732    fields.iter().map(|field| {
733        let field_name = field.name;
734        match field.kind {
735            FieldKind::Option | FieldKind::Positional => match &field.optionality {
736                Optionality::None => quote! {
737                    #field_name: #field_name.slot.unwrap()
738                },
739                Optionality::Optional | Optionality::Repeating => {
740                    quote! { #field_name: #field_name.slot }
741                }
742                Optionality::Defaulted(tokens) | Optionality::DefaultedRepeating(tokens) => {
743                    quote! {
744                        #field_name: #field_name.slot.unwrap_or_else(|| #tokens)
745                    }
746                }
747            },
748            FieldKind::Switch => field_name.into_token_stream(),
749            FieldKind::SubCommand => match field.optionality {
750                Optionality::None => quote! { #field_name: #field_name.unwrap() },
751                Optionality::Optional | Optionality::Repeating => field_name.into_token_stream(),
752                Optionality::Defaulted(_) | Optionality::DefaultedRepeating(_) => unreachable!(),
753            },
754        }
755    })
756}

758/// Declare a local slots to store each field in during parsing.
759///
760/// Most fields are stored in `Option<FieldType>` locals.
761/// `argh(option)` fields are stored in a `ParseValueSlotTy` along with a
762/// function that knows how to decode the appropriate value.
763fn declare_local_storage_for_redacted_fields<'a>(
764    fields: &'a [StructField<'a>],
765) -> impl Iterator<Item = TokenStream> + 'a {
766    fields.iter().map(|field| {
767        let field_name = &field.field.ident;
768
769        match field.kind {
770            FieldKind::Switch => {
771                quote! {
772                    let mut #field_name = argh::RedactFlag {
773                        slot: None,
774                    };
775                }
776            }
777            FieldKind::Option => {
778                let field_slot_type = match field.optionality {
779                    Optionality::Repeating => {
780                        quote! { std::vec::Vec<String> }
781                    }
782                    Optionality::DefaultedRepeating(_) => {
783                        quote! { std::option::Option<std::vec::Vec<String>> }
784                    }
785                    Optionality::None | Optionality::Optional | Optionality::Defaulted(_) => {
786                        quote! { std::option::Option<String> }
787                    }
788                };
789
790                quote! {
791                    let mut #field_name: argh::ParseValueSlotTy::<#field_slot_type, String> =
792                        argh::ParseValueSlotTy {
793                        slot: std::default::Default::default(),
794                        parse_func: |arg, _| { Ok(arg.to_owned()) },
795                    };
796                }
797            }
798            FieldKind::Positional => {
799                let field_slot_type = match field.optionality {
800                    Optionality::Repeating => {
801                        quote! { std::vec::Vec<String> }
802                    }
803                    Optionality::DefaultedRepeating(_) => {
804                        quote! { std::option::Option<std::vec::Vec<String>> }
805                    }
806                    Optionality::None | Optionality::Optional | Optionality::Defaulted(_) => {
807                        quote! { std::option::Option<String> }
808                    }
809                };
810
811                let arg_name = field.positional_arg_name();
812                quote! {
813                    let mut #field_name: argh::ParseValueSlotTy::<#field_slot_type, String> =
814                        argh::ParseValueSlotTy {
815                        slot: std::default::Default::default(),
816                        parse_func: |_, _| { Ok(#arg_name.to_owned()) },
817                    };
818                }
819            }
820            FieldKind::SubCommand => {
821                quote! { let mut #field_name: std::option::Option<std::vec::Vec<String>> = None; }
822            }
823        }
824    })
825}

827/// Unwrap non-optional fields and take options out of their tuple slots.
828fn unwrap_redacted_fields<'a>(
829    fields: &'a [StructField<'a>],
830) -> impl Iterator<Item = TokenStream> + 'a {
831    fields.iter().map(|field| {
832        let field_name = field.name;
833
834        match field.kind {
835            FieldKind::Switch => {
836                quote! {
837                    if let Some(__field_name) = #field_name.slot {
838                        __redacted.push(__field_name);
839                    }
840                }
841            }
842            FieldKind::Option => match field.optionality {
843                Optionality::Repeating => {
844                    quote! {
845                        __redacted.extend(#field_name.slot.into_iter());
846                    }
847                }
848                Optionality::DefaultedRepeating(_) => {
849                    quote! {
850                        if let Some(__field_name) = #field_name.slot {
851                            __redacted.extend(__field_name.into_iter());
852                        }
853                    }
854                }
855                Optionality::None | Optionality::Optional | Optionality::Defaulted(_) => {
856                    quote! {
857                        if let Some(__field_name) = #field_name.slot {
858                            __redacted.push(__field_name);
859                        }
860                    }
861                }
862            },
863            FieldKind::Positional => {
864                quote! {
865                    __redacted.extend(#field_name.slot.into_iter());
866                }
867            }
868            FieldKind::SubCommand => {
869                quote! {
870                    if let Some(__subcommand_args) = #field_name {
871                        __redacted.extend(__subcommand_args.into_iter());
872                    }
873                }
874            }
875        }
876    })
877}

879/// Entries of tokens like `("--some-flag-key", 5)` that map from a flag key string
880/// to an index in the output table.
881fn flag_str_to_output_table_map_entries<'a>(fields: &'a [StructField<'a>]) -> Vec<TokenStream> {
882    let mut flag_str_to_output_table_map = vec![];
883    for (i, (field, long_name)) in fields
884        .iter()
885        .filter_map(|field| field.long_name.as_ref().map(|long_name| (field, long_name)))
886        .enumerate()
887    {
888        if let Some(short) = &field.attrs.short {
889            let short = format!("-{}", short.value());
890            flag_str_to_output_table_map.push(quote! { (#short, #i) });
891        }
892
893        flag_str_to_output_table_map.push(quote! { (#long_name, #i) });
894    }
895    flag_str_to_output_table_map
896}

898/// For each non-optional field, add an entry to the `argh::MissingRequirements`.
899fn append_missing_requirements<'a>(
900    // missing_requirements_ident
901    mri: &syn::Ident,
902    fields: &'a [StructField<'a>],
903) -> impl Iterator<Item = TokenStream> + 'a {
904    let mri = mri.clone();
905    fields.iter().filter(|f| f.optionality.is_required()).map(move |field| {
906        let field_name = field.name;
907        match field.kind {
908            FieldKind::Switch => unreachable!("switches are always optional"),
909            FieldKind::Positional => {
910                let name = field.positional_arg_name();
911                quote! {
912                    if #field_name.slot.is_none() {
913                        #mri.missing_positional_arg(#name)
914                    }
915                }
916            }
917            FieldKind::Option => {
918                let name = field.long_name.as_ref().expect("options always have a long name");
919                quote! {
920                    if #field_name.slot.is_none() {
921                        #mri.missing_option(#name)
922                    }
923                }
924            }
925            FieldKind::SubCommand => {
926                let ty = field.ty_without_wrapper;
927                quote! {
928                    if #field_name.is_none() {
929                        #mri.missing_subcommands(
930                            <#ty as argh::SubCommands>::COMMANDS
931                                .iter()
932                                .cloned()
933                                .chain(
934                                    <#ty as argh::SubCommands>::dynamic_commands()
935                                        .iter()
936                                        .copied()
937                                ),
938                        )
939                    }
940                }
941            }
942        }
943    })
944}

946/// Require that a type can be a `switch`.
947/// Throws an error for all types except booleans and integers
948fn ty_expect_switch(errors: &Errors, ty: &syn::Type) -> bool {
949    fn ty_can_be_switch(ty: &syn::Type) -> bool {
950        if let syn::Type::Path(path) = ty {
951            if path.qself.is_some() {
952                return false;
953            }
954            if path.path.segments.len() != 1 {
955                return false;
956            }
957            let ident = &path.path.segments[0].ident;
958            // `Option<bool>` can be used as a `switch`.
959            if ident == "Option" {
960                if let PathArguments::AngleBracketed(args) = &path.path.segments[0].arguments {
961                    if let GenericArgument::Type(Type::Path(p)) = &args.args[0] {
962                        if p.path.segments[0].ident == "bool" {
963                            return true;
964                        }
965                    }
966                }
967            }
968            ["bool", "u8", "u16", "u32", "u64", "u128", "i8", "i16", "i32", "i64", "i128"]
969                .iter()
970                .any(|path| ident == path)
971        } else {
972            false
973        }
974    }
975
976    let res = ty_can_be_switch(ty);
977    if !res {
978        errors.err(ty, "switches must be of type `bool`, `Option<bool>`, or integer type");
979    }
980    res
981}

983/// Returns `Some(T)` if a type is `wrapper_name<T>` for any `wrapper_name` in `wrapper_names`.
984fn ty_inner<'a>(wrapper_names: &[&str], ty: &'a syn::Type) -> Option<&'a syn::Type> {
985    if let syn::Type::Path(path) = ty {
986        if path.qself.is_some() {
987            return None;
988        }
989        // Since we only check the last path segment, it isn't necessarily the case that
990        // we're referring to `std::vec::Vec` or `std::option::Option`, but there isn't
991        // a fool proof way to check these since name resolution happens after macro expansion,
992        // so this is likely "good enough" (so long as people don't have their own types called
993        // `Option` or `Vec` that take one generic parameter they're looking to parse).
994        let last_segment = path.path.segments.last()?;
995        if !wrapper_names.iter().any(|name| last_segment.ident == *name) {
996            return None;
997        }
998        if let syn::PathArguments::AngleBracketed(gen_args) = &last_segment.arguments {
999            let generic_arg = gen_args.args.first()?;
1000            if let syn::GenericArgument::Type(ty) = &generic_arg {
1001                return Some(ty);
1002            }
1003        }
1004    }
1005    None
1006}

1008/// Implements `FromArgs` and `SubCommands` for a `#![derive(FromArgs)]` enum.
1009fn impl_from_args_enum(
1010    errors: &Errors,
1011    name: &syn::Ident,
1012    type_attrs: &TypeAttrs,
1013    generic_args: &syn::Generics,
1014    de: &syn::DataEnum,
1015) -> TokenStream {
1016    parse_attrs::check_enum_type_attrs(errors, type_attrs, &de.enum_token.span);
1017
1018    // An enum variant like `<name>(<ty>)`
1019    struct SubCommandVariant<'a> {
1020        name: &'a syn::Ident,
1021        ty: &'a syn::Type,
1022    }
1023
1024    let mut dynamic_type_and_variant = None;
1025
1026    let variants: Vec<SubCommandVariant<'_>> = de
1027        .variants
1028        .iter()
1029        .filter_map(|variant| {
1030            let name = &variant.ident;
1031            let ty = enum_only_single_field_unnamed_variants(errors, &variant.fields)?;
1032            if parse_attrs::VariantAttrs::parse(errors, variant).is_dynamic.is_some() {
1033                if dynamic_type_and_variant.is_some() {
1034                    errors.err(variant, "Only one variant can have the `dynamic` attribute");
1035                }
1036                dynamic_type_and_variant = Some((ty, name));
1037                None
1038            } else {
1039                Some(SubCommandVariant { name, ty })
1040            }
1041        })
1042        .collect();
1043
1044    let name_repeating = std::iter::repeat(name.clone());
1045    let variant_ty = variants.iter().map(|x| x.ty).collect::<Vec<_>>();
1046    let variant_names = variants.iter().map(|x| x.name).collect::<Vec<_>>();
1047    let dynamic_from_args =
1048        dynamic_type_and_variant.as_ref().map(|(dynamic_type, dynamic_variant)| {
1049            quote! {
1050                if let Some(result) = <#dynamic_type as argh::DynamicSubCommand>::try_from_args(
1051                    command_name, args) {
1052                    return result.map(#name::#dynamic_variant);
1053                }
1054            }
1055        });
1056    let dynamic_redact_arg_values = dynamic_type_and_variant.as_ref().map(|(dynamic_type, _)| {
1057        quote! {
1058            if let Some(result) = <#dynamic_type as argh::DynamicSubCommand>::try_redact_arg_values(
1059                command_name, args) {
1060                return result;
1061            }
1062        }
1063    });
1064    let dynamic_commands = dynamic_type_and_variant.as_ref().map(|(dynamic_type, _)| {
1065        quote! {
1066            fn dynamic_commands() -> &'static [&'static argh::CommandInfo] {
1067                <#dynamic_type as argh::DynamicSubCommand>::commands()
1068            }
1069        }
1070    });
1071
1072    let (impl_generics, ty_generics, where_clause) = generic_args.split_for_impl();
1073    quote! {
1074        impl #impl_generics argh::FromArgs for #name #ty_generics #where_clause {
1075            fn from_args(command_name: &[&str], args: &[&str])
1076                -> std::result::Result<Self, argh::EarlyExit>
1077            {
1078                let subcommand_name = if let Some(subcommand_name) = command_name.last() {
1079                    *subcommand_name
1080                } else {
1081                    return Err(argh::EarlyExit::from("no subcommand name".to_owned()));
1082                };
1083
1084                #(
1085                    if subcommand_name == <#variant_ty as argh::SubCommand>::COMMAND.name {
1086                        return Ok(#name_repeating::#variant_names(
1087                            <#variant_ty as argh::FromArgs>::from_args(command_name, args)?
1088                        ));
1089                    }
1090                )*
1091
1092                #dynamic_from_args
1093
1094                Err(argh::EarlyExit::from("no subcommand matched".to_owned()))
1095            }
1096
1097            fn redact_arg_values(command_name: &[&str], args: &[&str]) -> std::result::Result<Vec<String>, argh::EarlyExit> {
1098                let subcommand_name = if let Some(subcommand_name) = command_name.last() {
1099                    *subcommand_name
1100                } else {
1101                    return Err(argh::EarlyExit::from("no subcommand name".to_owned()));
1102                };
1103
1104                #(
1105                    if subcommand_name == <#variant_ty as argh::SubCommand>::COMMAND.name {
1106                        return <#variant_ty as argh::FromArgs>::redact_arg_values(command_name, args);
1107                    }
1108                )*
1109
1110                #dynamic_redact_arg_values
1111
1112                Err(argh::EarlyExit::from("no subcommand matched".to_owned()))
1113            }
1114        }
1115
1116        impl #impl_generics argh::SubCommands for #name #ty_generics #where_clause {
1117            const COMMANDS: &'static [&'static argh::CommandInfo] = &[#(
1118                <#variant_ty as argh::SubCommand>::COMMAND,
1119            )*];
1120
1121            #dynamic_commands
1122        }
1123    }
1124}

1126/// Returns `Some(Bar)` if the field is a single-field unnamed variant like `Foo(Bar)`.
1127/// Otherwise, generates an error.
1128fn enum_only_single_field_unnamed_variants<'a>(
1129    errors: &Errors,
1130    variant_fields: &'a syn::Fields,
1131) -> Option<&'a syn::Type> {
1132    macro_rules! with_enum_suggestion {
1133        ($help_text:literal) => {
1134            concat!(
1135                $help_text,
1136                "\nInstead, use a variant with a single unnamed field for each subcommand:\n",
1137                "    enum MyCommandEnum {\n",
1138                "        SubCommandOne(SubCommandOne),\n",
1139                "        SubCommandTwo(SubCommandTwo),\n",
1140                "    }",
1141            )
1142        };
1143    }
1144
1145    match variant_fields {
1146        syn::Fields::Named(fields) => {
1147            errors.err(
1148                fields,
1149                with_enum_suggestion!(
1150                    "`#![derive(FromArgs)]` `enum`s do not support variants with named fields."
1151                ),
1152            );
1153            None
1154        }
1155        syn::Fields::Unit => {
1156            errors.err(
1157                variant_fields,
1158                with_enum_suggestion!(
1159                    "`#![derive(FromArgs)]` does not support `enum`s with no variants."
1160                ),
1161            );
1162            None
1163        }
1164        syn::Fields::Unnamed(fields) => {
1165            if fields.unnamed.len() != 1 {
1166                errors.err(
1167                    fields,
1168                    with_enum_suggestion!(
1169                        "`#![derive(FromArgs)]` `enum` variants must only contain one field."
1170                    ),
1171                );
1172                None
1173            } else {
1174                // `unwrap` is okay because of the length check above.
1175                let first_field = fields.unnamed.first().unwrap();
1176                Some(&first_field.ty)
1177            }
1178        }
1179    }
1180}
