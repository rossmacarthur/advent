6#[derive(Debug, Clone, Copy, Default, PartialEq)]
7struct Pixel {
8    value: char,
9    color: Option<Color>,
10}

12#[derive(Debug, Clone)]
13pub struct Image(Vec<Vec<Pixel>>);

15impl Pixel {
16    fn with_color(value: char, color: Color) -> Self {
17        Self {
18            value,
19            color: Some(color),
20        }
21    }
22}

24impl Image {
25    fn height(&self) -> usize {
26        self.0.len()
27    }

30impl fmt::Display for Pixel {
31    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
32        match (self.value, self.color) {
33            ('\x00', _) => {
34                write!(f, " ")
35            }
36            (v, Some(color)) => {
37                let v = Paint::new(v).fg(color);
38                write!(f, "{v}")
39            }
40            (v, None) => {
41                write!(f, "{v}")
42            }
43        }
44    }
45}

47impl fmt::Display for Image {
48    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
49        for row in &self.0 {
50            for pixel in row {
51                write!(f, "{pixel}")?;
52            }
53            writeln!(f)?;
54        }
55        Ok(())
56    }
57}

59fn pixels(s: &str, color: Color) -> Vec<Pixel> {
60    s.chars().map(|c| Pixel::with_color(c, color)).collect()
61}

63fn center(w: usize, pixels: &[Pixel]) -> Vec<Pixel> {
64    assert!(pixels.len() <= w);
65    let mut row = vec![Pixel::default(); w];
66    let n = (w - pixels.len()) / 2;
67    for (i, p) in pixels.iter().copied().enumerate() {
68        row[n + i] = p;
69    }
70    row
71}

73fn snowflakes(rng: &mut ThreadRng, w: usize, h: usize) -> Image {
74    let mut img = vec![vec![Pixel::default(); w]; h];
75    let mut i = 0;
76    while i < w * h {
77        img[i / w][i % w] = Pixel::with_color('·', Color::White);
78        i += rng.random_range(3..6);
79    }
80    Image(img)
81}

83fn tree(rng: &mut ThreadRng, mut w: usize) -> Image {
84    if w.is_multiple_of(2) {
85        w -= 1;
86    }
87
88    // Draw the tree
89    let mut img = vec![
90        center(w, &pixels(" * ", Color::Fixed(229))),
91        center(w, &pixels(" /_\\ ", Color::Green)),
92        center(w, &pixels(" /_\\_\\ ", Color::Green)),
93    ];
94    for i in 3..((w - 2) / 2) {
95        let left = format!(" /{} ", "_\\".repeat(i));
96        let right = format!(" {}\\ ", "/_".repeat(i));
97        img.push(center(w, &pixels(&left, Color::Green)));
98        img.push(center(w, &pixels(&right, Color::Green)));
99    }
100
101    // Add some baubles
102    let off = rng.random_range(0..3);
103    for (i, row) in img.iter_mut().skip(2).enumerate() {
104        let (left, mid) = row.split_at_mut(w / 3);
105        let (mid, right) = mid.split_at_mut(w / 3);
106
107        let mut colors = [Color::Fixed(229), Color::Fixed(69), Color::Fixed(204)];
108        colors.rotate_left((off + i) % 3);
109
110        for (section, color) in [left, mid, right].into_iter().zip(colors) {
111            if let Some(spot) = section.iter_mut().filter(|p| p.value == '_').choose(rng) {
112                *spot = Pixel::with_color('*', color);
113            }
114        }
115    }
116
117    // Finally add the pot
118    if w >= 15 {
119        let rim = pixels(" [_____] ", Color::Fixed(245));
120        img.push(center(w, &rim));
121        let bowl = pixels(" \\___/ ", Color::Fixed(245));
122        img.push(center(w, &bowl));
123    } else {
124        let n = if w >= 10 { 3 } else { 1 };
125        let bowl = pixels(&format!(" \\{}/ ", "_".repeat(n)), Color::Fixed(245));
126        img.push(center(w, &bowl));
127    }
128
129    Image(img)
130}

132fn paste(mut bg: Image, fg: Image, (x, y): (usize, usize)) -> Image {
133    let maxy = bg.height() - y;
134    for (i, row) in fg.0.into_iter().rev().enumerate() {
135        for (j, pixel) in row.into_iter().enumerate() {
136            if pixel != Pixel::default() {
137                bg.0[maxy - i - 1][x + j] = pixel;
138            }
139        }
140    }
141    bg
142}

144pub fn fun() -> Image {
145    let mut rng = rand::rng();
146    let width = 46;
147    let height = 19;
148
149    // Draw a background of snowflakes
150    let mut img = snowflakes(&mut rng, width, height);
151
152    // Generate the first medium sized tree
153    let mut x = 0;
154    let w = rng.random_range(11..=17);
155    img = paste(img, tree(&mut rng, w), (x, 0));
156    x += w;
157
158    // Generate a slightly bigger tree
159    let w = rng.random_range(13..=21);
160    img = paste(img, tree(&mut rng, w), (x, 0));
161    x += w;
162
163    // If there is enough space for another tree then generate it
164    let rem = width - x;
165    if rem >= 7 {
166        img = paste(img, tree(&mut rng, rem), (x, 0));
167    }
168
169    img
