5///

12/// In addition to the usual requirements of [`mem::transmute()`], the caller
13/// needs to ensure that the source and target types have the same size and
14/// alignment.
15#[inline]
16pub unsafe fn transmute_unchecked<A, B>(a: A) -> B {
17    let b = unsafe { ptr::read(&a as *const A as *const B) };
18    mem::forget(a);
